\rSec3[demo.cell]{Class template cell}

\begin{codeblock}
namespace std {
  template<class T>
  class @\libglobal{cell}@ {
  public:
    // \ref{demo.cell.cons}, constructors
    template<class U> explicit cell(U&& u);

    cell(const cell&)
      requires copyable_like<T>
    = default;

    // \ref{demo.cell.mod}, modifiers
    void assign(const T& v);

    void reset()
      requires resettable_like<T>;

    friend void swap(cell& a, cell& b);

  private:
    T @\exposidnc{value}@; // exposition only
  };
}
\end{codeblock}

\rSec4[demo.cell.cons]{Constructors}

\indexlibraryctor{cell}%
\begin{itemdecl}
template<class U>
  explicit cell(U&& u);
\end{itemdecl}

\begin{itemdescr}
\pnum
\constraints
\tcode{convertible_like<U, T>} is \tcode{true} and \tcode{convertible_like<U, cell<T>>} is \tcode{false}.

\pnum
\effects
Initializes the value with \tcode{u}.
\end{itemdescr}
