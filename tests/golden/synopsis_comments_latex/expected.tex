\rSec3[demo.tray]{Class tray}

\begin{codeblock}
namespace std {
  class @\libglobal{tray}@ {
  public:
    // \ref{demo.tray.cons}, constructors
    tray();

    // access functions
    // \ref{demo.tray.obs}, observers
    bool empty() const;
    int size() const; // see below
    /* the count of elements */
    int count(int) const;

    int capacity() const;

  private:
    int @\exposid{size}@ = 0; // \expos
  };
}
\end{codeblock}
