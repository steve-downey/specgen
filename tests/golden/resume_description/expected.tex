\rSec3[resume.assign]{Assignment}

\indexlibraryglobal{assign}%
\begin{itemdecl}
inline void assign();
\end{itemdecl}

\begin{itemdescr}
\pnum
\ensures
The stored value is one of:
\begin{itemize}
\item the argument, if a value was supplied; this continuation belongs to the first item.
\item the previous value, otherwise.
\end{itemize}
Otherwise, the object is unchanged.

\pnum
A separate postcondition paragraph.
\begin{itemize}
\item One further guarantee.
\end{itemize}
This continues the further guarantee.

\pnum
\remarks
A following element remains separate.
\end{itemdescr}
