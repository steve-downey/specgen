void reset();
\end{itemdecl}

\end{itemdescr}

\end{itemdescr}
\end{addedblock}

\indexlibrarymember{arm}{gadget}%
\begin{itemdecl}
void arm(int level);
\end{itemdecl}

\begin{itemdescr}
\pnum
\expects
\tcode{level} is \removed{positive.}\added{greater than zero.}

\pnum
\effects
Arms the gadget at \tcode{level}.

\begin{addedblock}
\pnum
Arming an armed gadget replaces its level.
\end{addedblock}
\end{itemdescr}
