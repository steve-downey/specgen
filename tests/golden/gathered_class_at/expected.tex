\rSec3[box.syn]{Header <box> synopsis}

\begin{codeblock}
struct @\libglobal{crate}@ {
  int size;
};

template<class T>
struct @\libglobal{pallet}@ {
  T slot;
};

struct @\libglobal{label}@ {
  int id;
};

struct @\libglobal{crumb}@ {
  int weight;
};

struct @\libglobal{strap}@ {
  int length;
};
\end{codeblock}

\pnum
A program that instantiates \tcode{crumb} is ill-formed unless \tcode{sizeof(short) >= 2} is \tcode{true}.

\begin{itemdescr}
\pnum
\remarks
A \tcode{crumb} is described where it is written.
\end{itemdescr}

\rSec4[box.crate]{Class `crate`}

\pnum
A program that instantiates \tcode{crate} is ill-formed unless \tcode{sizeof(int) >= 2} is \tcode{true}.

\begin{itemdescr}
\pnum
\remarks
A \tcode{crate} owns no storage.
\end{itemdescr}

\rSec4[box.pallet]{Class template `pallet`}

\pnum
A program that instantiates \tcode{pallet<T>} is ill-formed unless \tcode{sizeof(char) == 1} is \tcode{true}.

\rSec4[box.label]{Class `label`}

\begin{itemdescr}
\pnum
\remarks
A \tcode{label} names a \tcode{crate} and nothing else.
\end{itemdescr}

\rSec4[box.hinge]{Class `hinge`}

\begin{codeblock}
struct @\libglobal{hinge}@ {
  int pin;
};
\end{codeblock}

\rSec5[box.hinge.general]{General}

\pnum
A program that instantiates \tcode{hinge} is ill-formed unless \tcode{sizeof(int) >= 2} is \tcode{true}.

\begin{itemdescr}
\pnum
\remarks
A \tcode{hinge} joins two crates.
\end{itemdescr}
