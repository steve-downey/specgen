\rSec3[demo.views]{Range adaptor objects}

\indexlibraryglobal{adapt}%
\begin{itemdecl}
inline constexpr @\placeholder{unspecified}@ views::adapt;
\end{itemdecl}

\begin{itemdescr}
\pnum
\remarks
The name \tcode{views::adapt} denotes a range adaptor object. Given a subexpression \tcode{E}, the expression \tcode{views::adapt(E)} is expression-equivalent to \tcode{buffer(E)}.
\end{itemdescr}
