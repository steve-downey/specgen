\indexlibraryglobal{alias}%
\begin{itemdecl}

\end{itemdecl}

\begin{itemdescr}
\pnum
\returns
\tcode{value}.

\pnum
\remarks
A reference is written \tcode{int\&}.

\end{itemdescr}
