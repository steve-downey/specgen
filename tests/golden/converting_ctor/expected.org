#+begin_itemdecl
template<class U> constexpr explicit(@\seebelow@) optional(const optional<U>& rhs);
#+end_itemdecl

#+begin_pnum
/Constraints/:

- ~is_constructible_v<T, const U&>~ is ~true~,
- ~is_constructible_v<T, optional<U>&>~ is ~false~,
- ~is_constructible_v<T, optional<U>>~ is ~false~,
- ~is_convertible_v<optional<U>&, T>~ is ~false~.
#+end_pnum

#+begin_pnum
/Effects/: If ~rhs~ contains a value, direct-non-list-initializes the contained value with ~*rhs~.
#+end_pnum

#+begin_pnum
/Remarks/: The expression inside ~explicit~ is equivalent to ~!is_convertible_v<const U&, T>~, see ([[https://eel.is/c++draft/optional.general][@@html:[optional.general]@@@@latex:{[}optional.general{]}@@]]).
#+end_pnum
